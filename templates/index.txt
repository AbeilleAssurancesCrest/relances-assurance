<script>
let tousLesDossiers = [];

async function chargerDossiers() {
    let res = await fetch('/api/dossiers');
    tousLesDossiers = await res.json();
    afficherTableau(tousLesDossiers);
}

function afficherTableau(dossiers) {
    let html = '';

    dossiers.forEach(d => {
        let classRow = d.alerte_7j ? 'alerte-rouge' : '';
        let piecesBadge = d.pieces_manquantes.map(p => `<span class="badge bg-warning text-dark badge-piece">${p}</span>`).join(' ');
        let dateRelance = d.derniere_relance ? new Date(d.derniere_relance).toLocaleString('fr-FR') : 'Aucune';
        
        let badgeStatut = d.statut === 'Complet' 
            ? '<span class="badge bg-success">✅ Complet</span>' 
            : (d.alerte_7j ? '<span class="badge bg-danger">⚠️ Relance > 7j</span>' : '<span class="badge bg-secondary">En attente</span>');

        html += `
            <tr class="${classRow}">
                <td><strong>${d.num_contrat}</strong></td>
                <td><strong>${d.nom}</strong> ${d.prenom}</td>
                <td><small>${d.email}</small></td>
                <td>${d.date_effet}</td>
                <td>${piecesBadge || '<span class="text-success small">Toutes reçues !</span>'}</td>
                <td><small>${dateRelance}</small></td>
                <td>${badgeStatut}</td>
                <td class="text-center">
                    <button onclick="envoyerRelance(${d.id}, '${d.email}', '${d.nom} ${d.prenom}', ${JSON.stringify(d.pieces_manquantes).replace(/"/g, '&quot;')})" class="btn btn-primary btn-sm my-1" ${d.statut === 'Complet' ? 'disabled' : ''}>📧 Relancer</button>
                    
                    ${d.statut !== 'Complet' 
                        ? `<button onclick="changerStatut(${d.id}, 'Complet')" class="btn btn-success btn-sm my-1">✅ Valider</button>` 
                        : `<button onclick="changerStatut(${d.id}, 'En attente')" class="btn btn-outline-secondary btn-sm my-1">🔄 Rouvrir</button>`}
                    
                    <button onclick="voirHistorique(${d.id})" class="btn btn-info btn-sm text-white my-1">📜 Preuve</button>
                    <button onclick="supprimerDossier(${d.id})" class="btn btn-outline-danger btn-sm my-1">🗑️</button>
                </td>
            </tr>
        `;
    });
    document.getElementById('tableBody').innerHTML = html || '<tr><td colspan="8" class="text-center p-3 text-muted">Aucun dossier trouvé.</td></tr>';
}

function filtrerTableau() {
    let terme = document.getElementById('recherche').value.toLowerCase();
    let mefiltrer = tousLesDossiers.filter(d => 
        d.nom.toLowerCase().includes(terme) || 
        d.prenom.toLowerCase().includes(terme) || 
        d.num_contrat.toLowerCase().includes(terme) ||
        d.email.toLowerCase().includes(terme)
    );
    afficherTableau(mefiltrer);
}

async function ajouterDossier() {
    let pieces = Array.from(document.querySelectorAll('#listePieces input:checked')).map(c => c.value);
    let data = {
        num_contrat: document.getElementById('num_contrat').value,
        nom: document.getElementById('nom').value,
        prenom: document.getElementById('prenom').value,
        email: document.getElementById('email').value,
        date_effet: document.getElementById('date_effet').value,
        commentaire: document.getElementById('commentaire').value,
        pieces: pieces
    };

    await fetch('/api/dossiers', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify(data)
    });

    document.getElementById('formClient').reset();
    chargerDossiers();
}

async function changerStatut(id, statut) {
    await fetch('/api/statut', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({dossier_id: id, statut: statut})
    });
    chargerDossiers();
}

async function supprimerDossier(id) {
    if(confirm("Confirmer la suppression de ce dossier client ?")) {
        await fetch(`/api/dossiers/${id}`, { method: 'DELETE' });
        chargerDossiers();
    }
}

async function envoyerRelance(id, email, nomClient, pieces) {
    await fetch('/api/relancer', {
        method: 'POST',
        headers: {'Content-Type': 'application/json'},
        body: JSON.stringify({dossier_id: id, email: email, pieces: pieces})
    });

    let sujet = encodeURIComponent("Rappel : Pièces manquantes pour votre contrat d'assurance");
    let listeTexte = pieces.map(p => " - " + p).join("%0D%0A");
    let corps = encodeURIComponent(`Bonjour ${nomClient},%0D%0A%0D%0AAfin de finaliser votre dossier d'assurance, merci de bien vouloir nous faire parvenir les pièces suivantes :%0D%0A%0D%0A`) + listeTexte + encodeURIComponent(`%0D%0A%0D%0ACordialement,%0D%0AVotre Gestionnaire d'Assurances`);

    window.location.href = `mailto:${email}?subject=${sujet}&body=${corps}`;

    chargerDossiers();
}

async function voirHistorique(id) {
    let res = await fetch(`/api/historique/${id}`);
    let histo = await res.json();

    let content = '<ul class="list-group">';
    if(histo.length === 0) {
        content += '<li class="list-group-item">Aucune relance effectuée pour le moment.</li>';
    } else {
        histo.forEach(h => {
            let date = new Date(h.date_heure).toLocaleString('fr-FR');
            content += `
                <li class="list-group-item">
                    <strong>Date & Heure :</strong> ${date}<br>
                    <strong>Destinataire :</strong> ${h.email}<br>
                    <strong>Pièces réclamées :</strong> ${h.pieces.join(', ')}
                </li>
            `;
        });
    }
    content += '</ul>';

    document.getElementById('histoContent').innerHTML = content;
    new bootstrap.Modal(document.getElementById('modalHisto')).show();
}

chargerDossiers();
